1. Circle (Круг)
Объяснение для билета: Конструктор копирования и деструктор не требуются, т.к. в классе Circle нет указателей и не выделяется динамическая память (поля double x, y, r хранятся в стеке). Стандартный конструктор копирования (побитовое копирование) и деструктор по умолчанию работают корректно. 
#include <iostream>
#include <cmath>
using namespace std;

class Circle {
private:
    double x, y; // координаты центра
    double r;    // радиус

public:
    // --- Конструкторы ---
    // Конструктор с параметрами по умолчанию (заменяет конструктор по умолчанию и с параметрами)
    Circle(double _x = 0.0, double _y = 0.0, double _r = 1.0);

    // --- Обычные методы ---
    void Input();                                    // ввод с клавиатуры
    void Output() const;                             // вывод на экран
    double Area() const;                             // нахождение площади круга
    bool IsPointInside(double px, double py) const;  // проверка, попадает ли точка (px, py) в круг

    // --- Перегрузка операторов ---
    // Операции + и += (умножение круга на число: центр не меняется, радиус умножается на число)
    Circle operator+(double k) const;
    Circle& operator+=(double k);

    // Сравнение (== и !=) — два круга равны, если равны их радиусы
    bool operator==(const Circle& other) const;
    bool operator!=(const Circle& other) const;

    // Потоковый ввод >> и вывод << (дружественные функции)
    friend istream& operator>>(istream& in, Circle& c);
    friend ostream& operator<<(ostream& out, const Circle& c);
};

// ================= ОПРЕДЕЛЕНИЕ МЕТОДОВ ВНЕ КЛАССА =================

// Конструктор
Circle::Circle(double _x, double _y, double _r) : x(_x), y(_y) {
    r = (_r > 0) ? _r : 1.0; // радиус должен быть положительным
}

// Метод ввода
void Circle::Input() {
    cout << "Введите x, y и r: ";
    cin >> x >> y >> r;
}

// Метод вывода
void Circle::Output() const {
    cout << "Центр: (" << x << ", " << y << "), Радиус: " << r << endl;
}

// Нахождение площади (S = pi * r^2)
double Circle::Area() const {
    return 3.141592653589793 * r * r;
}

// Проверка попадания точки (px, py) в круг: (px - x)^2 + (py - y)^2 <= r^2
bool Circle::IsPointInside(double px, double py) const {
    return ((px - x) * (px - x) + (py - y) * (py - y)) <= (r * r);
}

// Оператор + (создается новый круг с радиусом r * k, текущий не меняется)
Circle Circle::operator+(double k) const {
    return Circle(x, y, r * k);
}

// Оператор += (изменяется радиус текущего круга)
Circle& Circle::operator+=(double k) {
    r *= k;
    return *this;
}

// Оператор == (два круга равны, если равны их радиусы)
bool Circle::operator==(const Circle& other) const {
    return r == other.r;
}

// Оператор != (два круга не равны, если не равны их радиусы)
bool Circle::operator!=(const Circle& other) const {
    return r != other.r;
}

// Потоковый ввод >> (БЕЗ слова friend и БЕЗ Circle::)
istream& operator>>(istream& in, Circle& c) {
    in >> c.x >> c.y >> c.r;
    return in;
}

// Потоковый вывод << (БЕЗ слова friend и БЕЗ Circle::)
ostream& operator<<(ostream& out, const Circle& c) {
    out << "Center(" << c.x << ", " << c.y << "), R = " << c.r;
    return out;
}
2. FreeVector (Вектор)
Объяснение для билета: Конструктор копирования и деструктор не требуются, т.к. в классе FreeVector нет указателей и не выделяется динамическая память (поля double x, y, z хранятся в стеке). Стандартный конструктор копирования и деструктор по умолчанию работают корректно. 
#include <iostream>
#include <cmath>
using namespace std;

class FreeVector {
private:
    double x, y, z; // координаты вектора

public:
    // --- Конструкторы ---
    FreeVector(double _x = 0.0, double _y = 0.0, double _z = 0.0);

    // --- Обычные методы ---
    void Input();               // ввод вектора
    void Output() const;        // вывод вектора
    double Length() const;      // нахождение длины (модуля) вектора

    // --- Перегрузка операторов ---
    // Операции + и += (сумма векторов)
    FreeVector operator+(const FreeVector& v) const;
    FreeVector& operator+=(const FreeVector& v);

    // Операции - и -= (разность векторов)
    FreeVector operator-(const FreeVector& v) const;
    FreeVector& operator-=(const FreeVector& v);

    // Операция * (скалярное произведение векторов — возвращает число double!)
    double operator*(const FreeVector& v) const;

    // Операции сравнения (== и !=)
    bool operator==(const FreeVector& v) const;
    bool operator!=(const FreeVector& v) const;

    // Потоковый ввод >> и вывод <<
    friend istream& operator>>(istream& in, FreeVector& v);
    friend ostream& operator<<(ostream& out, const FreeVector& v);
};

// ================= ОПРЕДЕЛЕНИЕ МЕТОДОВ ВНЕ КЛАССА =================

// Конструктор
FreeVector::FreeVector(double _x, double _y, double _z) : x(_x), y(_y), z(_z) {}

// Ввод
void FreeVector::Input() {
    cout << "Введите координаты вектора x, y, z: ";
    cin >> x >> y >> z;
}

// Вывод
void FreeVector::Output() const {
    cout << "(" << x << ", " << y << ", " << z << ")" << endl;
}

// Нахождение длины вектора: sqrt(x^2 + y^2 + z^2)
double FreeVector::Length() const {
    return sqrt(x * x + y * y + z * z);
}

// Сумма векторов (+) — создает новый вектор
FreeVector FreeVector::operator+(const FreeVector& v) const {
    return FreeVector(x + v.x, y + v.y, z + v.z);
}

// Сумма векторов (+=) — изменяет текущий вектор
FreeVector& FreeVector::operator+=(const FreeVector& v) {
    x += v.x;
    y += v.y;
    z += v.z;
    return *this;
}

// Разность векторов (-) — создает новый вектор
FreeVector FreeVector::operator-(const FreeVector& v) const {
    return FreeVector(x - v.x, y - v.y, z - v.z);
}

// Разность векторов (-=) — изменяет текущий вектор
FreeVector& FreeVector::operator-=(const FreeVector& v) {
    x -= v.x;
    y -= v.y;
    z -= v.z;
    return *this;
}

// Скалярное произведение векторов (*)
double FreeVector::operator*(const FreeVector& v) const {
    return x * v.x + y * v.y + z * v.z;
}

// Сравнение на равенство (==)
bool FreeVector::operator==(const FreeVector& v) const {
    return (x == v.x) && (y == v.y) && (z == v.z);
}

// Сравнение на неравенство (!=)
bool FreeVector::operator!=(const FreeVector& v) const {
    return !(*this == v);
}

// Потоковый ввод >>
istream& operator>>(istream& in, FreeVector& v) {
    in >> v.x >> v.y >> v.z;
    return in;
}

// Потоковый вывод <<
ostream& operator<<(ostream& out, const FreeVector& v) {
    out << "(" << v.x << ", " << v.y << ", " << v.z << ")";
    return out;
}
3. Complex (Комплексное)
Объяснение для билета: Конструктор копирования и деструктор не требуются, т.к. член-данные (double x, y) не располагаются в динамической памяти. Достаточно стандартного конструктора копирования и деструктора по умолчанию. 
#include <iostream>
#include <cmath>
using namespace std;

class Complex {
private:
    double x; // действительная часть (Re)
    double y; // мнимая часть (Im)

public:
    // --- Конструкторы ---
    Complex(double _x = 0.0, double _y = 0.0);

    // --- Обычные методы ---
    void Input();                   // ввод
    void Output() const;            // вывод
    double Modulus() const;         // модуль комплексного числа
    Complex Conjugate() const;      // комплексно-сопряженное число (x - i*y)

    // --- Перегрузка арифметических операторов ---
    Complex operator+(const Complex& c) const;
    Complex& operator+=(const Complex& c);

    Complex operator-(const Complex& c) const;
    Complex& operator-=(const Complex& c);

    Complex operator*(const Complex& c) const;
    Complex& operator*=(const Complex& c);

    Complex operator/(const Complex& c) const;
    Complex& operator/=(const Complex& c);

    // --- Операции сравнения ---
    bool operator==(const Complex& c) const;
    bool operator!=(const Complex& c) const;

    // --- Потоковый ввод >> и вывод << ---
    friend istream& operator>>(istream& in, Complex& c);
    friend ostream& operator<<(ostream& out, const Complex& c);
};

// ================= ОПРЕДЕЛЕНИЕ МЕТОДОВ ВНЕ КЛАССА =================

// Конструктор
Complex::Complex(double _x, double _y) : x(_x), y(_y) {}

// Ввод
void Complex::Input() {
    cout << "Введите действительную и мнимую часть (x, y): ";
    cin >> x >> y;
}

// Вывод
void Complex::Output() const {
    if (y >= 0)
        cout << x << " + " << y << "i" << endl;
    else
        cout << x << " - " << -y << "i" << endl;
}

// Модуль комплексного числа: sqrt(x^2 + y^2)
double Complex::Modulus() const {
    return sqrt(x * x + y * y);
}

// Комплексно-сопряженное число (меняется знак мнимой части)
Complex Complex::Conjugate() const {
    return Complex(x, -y);
}

// Сложение (+)
Complex Complex::operator+(const Complex& c) const {
    return Complex(x + c.x, y + c.y);
}

// Сложение (+=)
Complex& Complex::operator+=(const Complex& c) {
    x += c.x;
    y += c.y;
    return *this;
}

// Вычитание (-)
Complex Complex::operator-(const Complex& c) const {
    return Complex(x - c.x, y - c.y);
}

// Вычитание (-=)
Complex& Complex::operator-=(const Complex& c) {
    x -= c.x;
    y -= c.y;
    return *this;
}

// Умножение (*): (x1 + i*y1)*(x2 + i*y2) = (x1*x2 - y1*y2) + i*(x1*y2 + y1*x2)
Complex Complex::operator*(const Complex& c) const {
    return Complex(x * c.x - y * c.y, x * c.y + y * c.x);
}

// Умножение (*=)
Complex& Complex::operator*=(const Complex& c) {
    double tempX = x * c.x - y * c.y;
    y = x * c.y + y * c.x;
    x = tempX;
    return *this;
}

// Деление (/): домножение числителя и знаменателя на сопряженное к делителю
Complex Complex::operator/(const Complex& c) const {
    double denom = c.x * c.x + c.y * c.y; // знаменатель: x2^2 + y2^2
    return Complex((x * c.x + y * c.y) / denom, (y * c.x - x * c.y) / denom);
}

// Деление (/=)
Complex& Complex::operator/=(const Complex& c) {
    double denom = c.x * c.x + c.y * c.y;
    double tempX = (x * c.x + y * c.y) / denom;
    y = (y * c.x - x * c.y) / denom;
    x = tempX;
    return *this;
}

// Сравнение (==)
bool Complex::operator==(const Complex& c) const {
    return (x == c.x) && (y == c.y);
}

// Сравнение (!=)
bool Complex::operator!=(const Complex& c) const {
    return !(*this == c);
}

// Потоковый ввод >>
istream& operator>>(istream& in, Complex& c) {
    in >> c.x >> c.y;
    return in;
}

// Потоковый вывод <<
ostream& operator<<(ostream& out, const Complex& c) {
    if (c.y >= 0)
        out << c.x << "+" << c.y << "i";
    else
        out << c.x << c.y << "i";
    return out;
}
4. Array (Массив — все 17 п.) 
#include <iostream>
#include <cstdlib>
using namespace std;

class Array {
private:
    int* arr; // указатель на динамический массив
    int n;    // число элементов массива

public:
    // --- Конструкторы и деструктор ---
    Array(int n_ = 0);                      // конструктор по умолчанию / с размером
    Array(const int* a, int n_);            // перевод обычного массива int* в объект Array
    Array(const Array& other);              // конструктор копирования (ОБЯЗАТЕЛЕН, т.к. динамическая память!)
    ~Array();                               // деструктор (ОБЯЗАТЕЛЕН!)

    // --- Методы из задания ---
    void Input();                           // функция ввода
    void Output() const;                    // функция вывода
    void Random(int minVal = 0, int maxVal = 100); // формирование случайного массива

    int FindKey(int key) const;             // поиск ключа (возвращает индекс или -1)
    int MaxIndex() const;                   // нахождение max элемента (возвращает индекс)
    int MinIndex() const;                   // нахождение min элемента (возвращает индекс)

    // Добавление ключа на позицию pos:
    void AddPos(int key, int pos);          // изменяется текущий объект
    Array AddPosNew(int key, int pos) const;// создается новый объект

    // Удаление элемента с позиции pos:
    void DelPos(int pos);                   // изменяется текущий объект
    Array DelPosNew(int pos) const;         // создается новый объект

    // --- Перегрузка операторов ---
    Array& operator=(const Array& other);   // = (присвоение)
    int& operator[](int index);             // [] (возвращает ссылку)

    Array operator+(int key) const;         // +(int key) добавление ключа в конец (новый объект)
    Array& operator+=(int key);             // +=(int key) добавление ключа в конец (текущий объект)

    Array operator+(const Array& other) const; // +(Array) добавление массива в конец (новый объект)
    Array& operator+=(const Array& other);     // +=(Array) добавление массива в конец (текущий объект)

    Array operator-(int key) const;         // -(int key) удаление первого вхождения key (новый объект)
    Array& operator-=(int key);             // -=(int key) удаление первого вхождения key (текущий объект)

    bool operator==(const Array& other) const; // == (сравнение на равенство)
    bool operator!=(const Array& other) const; // != (сравнение на неравенство)

    // Потоковый ввод >> и вывод <<
    friend istream& operator>>(istream& in, Array& a);
    friend ostream& operator<<(ostream& out, const Array& a);
};

// ================= ОПРЕДЕЛЕНИЕ МЕТОДОВ ВНЕ КЛАССА =================

// 1. Конструктор с размером по умолчанию
Array::Array(int n_) {
    n = (n_ > 0) ? n_ : 0;
    if (n > 0) {
        arr = new int[n];
        for (int i = 0; i < n; i++) arr[i] = 0;
    } else {
        arr = nullptr;
    }
}

// 2. Конструктор перевода обычного массива в объект Array
Array::Array(const int* a, int n_) {
    n = (n_ > 0) ? n_ : 0;
    if (n > 0) {
        arr = new int[n];
        for (int i = 0; i < n; i++) arr[i] = a[i];
    } else {
        arr = nullptr;
    }
}

// 3. Конструктор копирования (глубокое копирование)
Array::Array(const Array& other) {
    n = other.n;
    if (n > 0) {
        arr = new int[n];
        for (int i = 0; i < n; i++) arr[i] = other.arr[i];
    } else {
        arr = nullptr;
    }
}

// 4. Деструктор
Array::~Array() {
    delete[] arr;
}

// 5. Ввод и вывод
void Array::Input() {
    for (int i = 0; i < n; i++) cin >> arr[i];
}

void Array::Output() const {
    for (int i = 0; i < n; i++) cout << arr[i] << " ";
    cout << endl;
}

// 6. Формирование случайного массива
void Array::Random(int minVal, int maxVal) {
    for (int i = 0; i < n; i++) {
        arr[i] = minVal + rand() % (maxVal - minVal + 1);
    }
}

// 7. Функция поиска ключа (возвращает индекс первого вхождения или -1)
int Array::FindKey(int key) const {
    for (int i = 0; i < n; i++) {
        if (arr[i] == key) return i;
    }
    return -1;
}

// 8. Нахождение индекса максимального и минимального элемента
int Array::MaxIndex() const {
    if (n == 0) return -1;
    int maxIdx = 0;
    for (int i = 1; i < n; i++) {
        if (arr[i] > arr[maxIdx]) maxIdx = i;
    }
    return maxIdx;
}

int Array::MinIndex() const {
    if (n == 0) return -1;
    int minIdx = 0;
    for (int i = 1; i < n; i++) {
        if (arr[i] < arr[minIdx]) minIdx = i;
    }
    return minIdx;
}

// 9. Добавление ключа key на позицию pos (изменяется текущий объект)
void Array::AddPos(int key, int pos) {
    if (pos < 0 || pos > n) return;
    int* temp = new int[n + 1];
    for (int i = 0; i < pos; i++) temp[i] = arr[i];
    temp[pos] = key;
    for (int i = pos; i < n; i++) temp[i + 1] = arr[i];
    delete[] arr;
    arr = temp;
    n++;
}

// Добавление ключа key на позицию pos (создается новый объект)
Array Array::AddPosNew(int key, int pos) const {
    Array res(*this);
    res.AddPos(key, pos);
    return res;
}

// 10. Удаление элемента с позиции pos (изменяется текущий объект)
void Array::DelPos(int pos) {
    if (pos < 0 || pos >= n) return;
    if (n == 1) {
        delete[] arr;
        arr = nullptr;
        n = 0;
        return;
    }
    int* temp = new int[n - 1];
    int j = 0;
    for (int i = 0; i < n; i++) {
        if (i != pos) temp[j++] = arr[i];
    }
    delete[] arr;
    arr = temp;
    n--;
}

// Удаление элемента с позиции pos (создается новый объект, текущий не меняется)
Array Array::DelPosNew(int pos) const {
    if (pos < 0 || pos >= n) return *this;
    Array res(n - 1);
    int j = 0;
    for (int i = 0; i < n; i++) {
        if (i != pos) res.arr[j++] = arr[i];
    }
    return res;
}

// 11. Оператор присвоения (=)
Array& Array::operator=(const Array& other) {
    if (this != &other) { // обязательно проверка на самоприсваивание!
        delete[] arr;     // освобождаем старую память
        n = other.n;
        if (n > 0) {
            arr = new int[n];
            for (int i = 0; i < n; i++) arr[i] = other.arr[i];
        } else {
            arr = nullptr;
        }
    }
    return *this;
}

// 12. Оператор индексации ([]) — возвращает ссылку
int& Array::operator[](int index) {
    return arr[index];
}

// 13. Операторы += и + (добавление ключа key в конец массива)
Array& Array::operator+=(int key) {
    int* temp = new int[n + 1];
    for (int i = 0; i < n; i++) temp[i] = arr[i];
    temp[n] = key;
    delete[] arr;
    arr = temp;
    n++;
    return *this;
}

Array Array::operator+(int key) const {
    Array res(*this);
    res += key;
    return res;
}

// 14. Операторы += и + (добавление другого массива Array в конец массива)
Array& Array::operator+=(const Array& other) {
    if (other.n == 0) return *this;
    int* temp = new int[n + other.n];
    for (int i = 0; i < n; i++) temp[i] = arr[i];
    for (int i = 0; i < other.n; i++) temp[n + i] = other.arr[i];
    delete[] arr;
    arr = temp;
    n += other.n;
    return *this;
}

Array Array::operator+(const Array& other) const {
    Array res(*this);
    res += other;
    return res;
}

// 15. Операторы -= и - (удаление первого вхождения ключа key)
Array& Array::operator-=(int key) {
    int pos = FindKey(key); // ищем индекс первого вхождения
    if (pos != -1) {
        DelPos(pos);        // если нашли — удаляем элемент на позиции pos
    }
    return *this;
}

Array Array::operator-(int key) const {
    Array res(*this);
    res -= key;
    return res;
}

// 16. Операторы сравнения (== и !=)
bool Array::operator==(const Array& other) const {
    if (n != other.n) return false;
    for (int i = 0; i < n; i++) {
        if (arr[i] != other.arr[i]) return false;
    }
    return true;
}

bool Array::operator!=(const Array& other) const {
    return !(*this == other);
}

// 17. Потоковый ввод >> и вывод <<
istream& operator>>(istream& in, Array& a) {
    for (int i = 0; i < a.n; i++) in >> a.arr[i];
    return in;
}

ostream& operator<<(ostream& out, const Array& a) {
    for (int i = 0; i < a.n; i++) out << a.arr[i] << " ";
    return out;
}

#include <iostream>
#include <cstring>
using namespace std;

class String {
private:
    char* line; // указатель на строку (динамический символьный массив)
    int len;    // длина строки (без учета нуль-терминатора '\0')

public:

    String(int L = 80);               // конструктор по умолчанию / заданной длины
    String(const char* s);            // конструктор инициализации C-строкой
    String(const String& other);      // конструктор копирования
    ~String();                        // деструктор

    // --- Перегрузка операторов ---

    char& operator[](int index);                 // [] (возвращает ссылку на символ)

    String operator+(const String& other) const; // + (конкатенация — создает новую строку)
    String& operator+=(const String& other);     // += (конкатенация — изменяет текущую строку)

    bool operator!=(const String& other) const;  // != (сравнение строк на неравенство)

}

    line = new char[len + 1];
    strcpy(line, s);
}

String::String(const String& other) {
    len = other.len;
    line = new char[len + 1];
    strcpy(line, other.line);
}

String::~String() {
    delete[] line;
}

String& String::operator=(const String& other) {
    if (this != &other) { // проверка на самоприсваивание
        delete[] line;
        len = other.len;
        line = new char[len + 1];
        strcpy(line, other.line);
    }
    return *this;
}

// 6. Оператор индексации ([]) — возвращает ссылку
char& String::operator[](int index) {
    return line[index];
}

// 7. Оператор + (конкатенация / сцепление строк, создается новый объект)
String String::operator+(const String& other) const {
    String res(len + other.len);
    strcpy(res.line, line);
    strcat(res.line, other.line);
    return res;
}

// 8. Оператор += (конкатенация, изменяется текущий объект)
String& String::operator+=(const String& other) {
    len += other.len;
    char* temp = new char[len + 1];
    strcpy(temp, line);
    strcat(temp, other.line);
    delete[] line;
    line = temp;
    return *this;
}

    return strcmp(line, other.line) == 0;
}

    return strcmp(line, other.line) != 0;
}

    in >> buffer; // или in.getline(buffer, 256);
    delete[] s.line;
    s.len = strlen(buffer);
    s.line = new char[s.len + 1];

6. Stack<T> (Шаблон стека) 
#include <iostream>
using namespace std;

template <class T>
class Stack {
private:
    T* arr;  // массив элементов стека
    int max; // максимальный размер массива arr
    int n;   // количество реальных элементов в стеке (индекс вершины)

public:
    // --- Конструкторы и деструктор ---
    Stack(int k = 50);                  // конструктор (по умолчанию k = 50)
    Stack(const Stack<T>& other);       // конструктор копирования
    ~Stack();                           // деструктор

    // --- Методы работы со стеком ---
    bool IsEmpty() const;               // проверка на пустоту
    bool IsFull() const;                // проверка на заполненность
    bool Push(const T& item);           // добавить в стек элемент (на вершину)
    bool Pop(T& item);                  // удалить из стека элемент (с сохранением в item)
    T Top() const;                      // взять значение с вершины стека (без удаления)

    // --- Перегрузка операторов ---
    Stack<T>& operator=(const Stack<T>& other); // = (присвоение)

    // Потоковый вывод <<
    template <class U>
    friend ostream& operator<<(ostream& out, const Stack<U>& s);
};

// ================= ОПРЕДЕЛЕНИЕ МЕТОДОВ ШАБЛОНА ВНЕ КЛАССА =================

// 1. Конструктор
template <class T>
Stack<T>::Stack(int k) {
    max = (k > 0) ? k : 50;
    n = 0; // изначально стек пуст
    arr = new T[max];
}

// 2. Конструктор копирования
template <class T>
Stack<T>::Stack(const Stack<T>& other) {
    max = other.max;
    n = other.n;
    arr = new T[max];
    for (int i = 0; i < n; i++) {
        arr[i] = other.arr[i];
    }
}

// 3. Деструктор
template <class T>
Stack<T>::~Stack() {
    delete[] arr;
}

// 4. Проверка на пустоту
template <class T>
bool Stack<T>::IsEmpty() const {
    return n == 0;
}

// 5. Проверка на заполненность
template <class T>
bool Stack<T>::IsFull() const {
    return n == max;
}

// 6. Добавить в стек элемент (Push)
template <class T>
bool Stack<T>::Push(const T& item) {
    if (IsFull()) return false; // если стек полон, добавить нельзя
    arr[n++] = item;            // кладем элемент и увеличиваем счетчик n
    return true;
}

// 7. Удалить из стека элемент (Pop)
template <class T>
bool Stack<T>::Pop(T& item) {
    if (IsEmpty()) return false; // если стек пуст, удалять нечего
    item = arr[--n];             // уменьшаем n и забираем верхний элемент
    return true;
}

// 8. Взять значение с вершины стека (Top)
template <class T>
T Stack<T>::Top() const {
    return arr[n - 1];
}

// 9. Оператор присвоения (=)
template <class T>
Stack<T>& Stack<T>::operator=(const Stack<T>& other) {
    if (this != &other) {
        delete[] arr;
        max = other.max;
        n = other.n;
        arr = new T[max];
        for (int i = 0; i < n; i++) {
            arr[i] = other.arr[i];
        }
    }
    return *this;
}

// 10. Потоковый вывод <<
template <class U>
ostream& operator<<(ostream& out, const Stack<U>& s) {
    for (int i = 0; i < s.n; i++) {
        out << s.arr[i] << " ";
    }
    return out;
}
