1. Circle (Круг)
Почему не нужны констр. копирования и деструктор: Нет указателей и динамической памяти (поля double x, y, r лежат в стеке). Стандартного побитового копирования и деструктора компилятора достаточно.
Объявление класса Circle
#include <iostream>
#include <cmath>
using namespace std;

class Circle {
private:
  double x, y; // координаты центра
  double r;    // радиус

public:
  // Конструктор (заменяет по умолч. и с параметрами)
  Circle(double _x = 0.0, double _y = 0.0, double _r = 1.0);

  // Обычные методы
  void Input();
  void Output() const;
  double Area() const;
  bool IsPointInside(double px, double py) const;

  // Операции + и += (умножение радиуса на число)
  Circle operator+(double k) const;
  Circle& operator+=(double k);

  // Сравнение (== и !=) по радиусу
  bool operator==(const Circle& other) const;
  bool operator!=(const Circle& other) const;

  // Потоковый ввод >> и вывод <<
  friend istream& operator>>(istream& in, Circle& c);
  friend ostream& operator<<(ostream& out, const Circle& c);
};
Методы вне класса
// Конструктор
Circle::Circle(double _x, double _y, double _r) : x(_x), y(_y) {
  r = (_r > 0) ? _r : 1.0;
}

// Ввод
void Circle::Input() {
  cout << "Введите x, y и r: ";
  cin >> x >> y >> r;
}

// Вывод
void Circle::Output() const {
  cout << "Центр: (" << x << ", " << y << "), R: " << r << endl;
}

// Нахождение площади (S = pi * r^2)
double Circle::Area() const {
  return 3.141592653589793 * r * r;
}

// Попадание точки (px, py) в круг
bool Circle::IsPointInside(double px, double py) const {
  return ((px - x)*(px - x) + (py - y)*(py - y)) <= (r * r);
}
Перегрузка операторов
// Оператор + (новый круг с r * k)
Circle Circle::operator+(double k) const {
  return Circle(x, y, r * k);
}

// Оператор += (меняет текущий круг)
Circle& Circle::operator+=(double k) {
  r *= k;
  return *this;
}

// Оператор == (равны, если равны радиусы)
bool Circle::operator==(const Circle& other) const {
  return r == other.r;
}

// Оператор !=
bool Circle::operator!=(const Circle& other) const {
  return r != other.r;
}

// Потоковый ввод >> (БЕЗ friend и БЕЗ Circle::)
istream& operator>>(istream& in, Circle& c) {
  in >> c.x >> c.y >> c.r;
  return in;
}

// Потоковый вывод << (БЕЗ friend и БЕЗ Circle::)
ostream& operator<<(ostream& out, const Circle& c) {
  out << "Center(" << c.x << ", " << c.y << "), R = " << c.r;
  return out;
}
2. FreeVector (Вектор)
Почему не нужны констр. копирования и деструктор: Нет динамической памяти и указателей (поля double x, y, z). Стандартный конструктор копирования и деструктор работают корректно.
Объявление класса FreeVector
#include <iostream>
#include <cmath>
using namespace std;

class FreeVector {
private:
  double x, y, z; // координаты вектора

public:
  FreeVector(double _x = 0.0, double _y = 0.0, double _z = 0.0);

  void Input();
  void Output() const;
  double Length() const; // длина вектора

  // Сумма векторов (+ и +=)
  FreeVector operator+(const FreeVector& v) const;
  FreeVector& operator+=(const FreeVector& v);

  // Разность векторов (- и -=)
  FreeVector operator-(const FreeVector& v) const;
  FreeVector& operator-=(const FreeVector& v);

  // Скалярное произведение (*) -> double!
  double operator*(const FreeVector& v) const;

  // Сравнение (== и !=)
  bool operator==(const FreeVector& v) const;
  bool operator!=(const FreeVector& v) const;

  // Потоковый ввод >> и вывод <<
  friend istream& operator>>(istream& in, FreeVector& v);
  friend ostream& operator<<(ostream& out, const FreeVector& v);
};
Методы и операторы вне класса
// Конструктор
FreeVector::FreeVector(double _x, double _y, double _z)

// Ввод
void FreeVector::Input() {

}

// Вывод

}

  return sqrt(x * x + y * y + z * z);
}

  return FreeVector(x + v.x, y + v.y, z + v.z);
}

// Сумма (+=)

  x += v.x; y += v.y; z += v.z;
  return *this;

  return FreeVector(x - v.x, y - v.y, z - v.z);
}

  x -= v.x; y -= v.y; z -= v.z;
  return *this;
}

// Скалярное произведение (*)

  return x * v.x + y * v.y + z * v.z;
}

  return (x == v.x) && (y == v.y) && (z == v.z);

bool FreeVector::operator!=(const FreeVector& v) const {
  return !(*this == v);
}

// Потоковый ввод >>
istream& operator>>(istream& in, FreeVector& v) {
  in >> v.x >> v.y >> v.z;
  return in;
}

// Потоковый вывод <<
ostream& operator<<(ostream& out, const FreeVector& v) {
  out << "(" << v.x << ", " << v.y << ", " << v.z << ")";
  return out;
}
3. Complex (Комплексное)
Почему не нужны констр. копирования и деструктор: Поля double x, y не находятся в динамической памяти. Достаточно стандартного конструктора копирования и деструктора по умолчанию.

#include <iostream>
#include <cmath>
using namespace std;

class Complex {

  void Output() const;
  double Modulus() const;    // модуль

  Complex& operator+=(const Complex& c);

  Complex operator-(const Complex& c) const;
  Complex& operator-=(const Complex& c);

  Complex operator*(const Complex& c) const;
  Complex& operator*=(const Complex& c);

  Complex operator/(const Complex& c) const;
  Complex& operator/=(const Complex& c);

  // Сравнение
  bool operator==(const Complex& c) const;
  bool operator!=(const Complex& c) const;

  // Потоки >> и <<
  friend istream& operator>>(istream& in, Complex& c);
  friend ostream& operator<<(ostream& out, const Complex& c);
};
Методы и операторы вне класса
// Конструктор
Complex::Complex(double _x, double _y) : x(_x), y(_y) {}

void Complex::Input() { cin >> x >> y; }

void Complex::Output() const {
  if (y >= 0) cout << x << " + " << y << "i" << endl;
  else cout << x << " - " << -y << "i" << endl;
}

// Модуль: sqrt(x^2 + y^2)
double Complex::Modulus() const {
  return sqrt(x * x + y * y);
}

// Комплексно-сопряженное
Complex Complex::Conjugate() const {
  return Complex(x, -y);
}

// Сложение (+ и +=)
Complex Complex::operator+(const Complex& c) const {
  return Complex(x + c.x, y + c.y);
}
Complex& Complex::operator+=(const Complex& c) {
  x += c.x; y += c.y;
  return *this;
}

// Вычитание (- и -=)
Complex Complex::operator-(const Complex& c) const {
  return Complex(x - c.x, y - c.y);
}
Complex& Complex::operator-=(const Complex& c) {
  x -= c.x; y -= c.y;
  return *this;
}

// Умножение (* и *=)
Complex Complex::operator*(const Complex& c) const {
  return Complex(x*c.x - y*c.y, x*c.y + y*c.x);
}
Complex& Complex::operator*=(const Complex& c) {
  double tempX = x*c.x - y*c.y;
  y = x*c.y + y*c.x;
  x = tempX;
  return *this;
}

// Деление (/ и /=)
Complex Complex::operator/(const Complex& c) const {
  double d = c.x*c.x + c.y*c.y;
  return Complex((x*c.x + y*c.y)/d, (y*c.x - x*c.y)/d);
}
Complex& Complex::operator/=(const Complex& c) {
  double d = c.x*c.x + c.y*c.y;
  double tempX = (x*c.x + y*c.y)/d;
  y = (y*c.x - x*c.y)/d;
  x = tempX;
  return *this;
}

// Сравнение (== и !=)
bool Complex::operator==(const Complex& c) const {
  return (x == c.x) && (y == c.y);
}
bool Complex::operator!=(const Complex& c) const {
  return !(*this == c);
}

// Потоковый ввод >> и вывод <<
istream& operator>>(istream& in, Complex& c) {
  in >> c.x >> c.y;
  return in;
}
ostream& operator<<(ostream& out, const Complex& c) {
  if (c.y >= 0) out << c.x << "+" << c.y << "i";
  else out << c.x << c.y << "i";
  return out;
}
4. Array (Массив - все 17 п.)

#include <cstdlib>
using namespace std;

class Array {
private:
  int* arr; // указатель на динамический массив
  int n;    // число элементов массива

5-8. Ввод/Вывод, Random, FindKey, Max/Min
// 5. Ввод и вывод
void Array::Input() {
  for (int i = 0; i < n; i++) cin >> arr[i];
}
void Array::Output() const {
  for (int i = 0; i < n; i++) cout << arr[i] << " ";

// 6. Случайный массив
void Array::Random(int minVal, int maxVal) {
  for (int i = 0; i < n; i++)
    arr[i] = minVal + rand() % (maxVal - minVal + 1);
}

// 7. Поиск ключа (индекс или -1)
int Array::FindKey(int key) const {
  for (int i = 0; i < n; i++) {
    if (arr[i] == key) return i;
  }
  return -1;
}

// 8. Индекс max и min элемента
int Array::MaxIndex() const {
  if (n == 0) return -1;
  int maxIdx = 0;
  for (int i = 1; i < n; i++)
    if (arr[i] > arr[maxIdx]) maxIdx = i;
  return maxIdx;
}

int Array::MinIndex() const {
  if (n == 0) return -1;
  int minIdx = 0;
  for (int i = 1; i < n; i++)
    if (arr[i] < arr[minIdx]) minIdx = i;
  return minIdx;
}
9-10. Добавление/Удаление позиции (AddPos / DelPos)
// 9. Добавление ключа на позицию pos (меняет текущий)
void Array::AddPos(int key, int pos) {
  if (pos < 0 || pos > n) return;
  int* temp = new int[n + 1];
  for (int i = 0; i < pos; i++) temp[i] = arr[i];
  temp[pos] = key;
  for (int i = pos; i < n; i++) temp[i + 1] = arr[i];
  delete[] arr;
  arr = temp;
  n++;
}

// Добавление на позицию pos (новый объект)
Array Array::AddPosNew(int key, int pos) const {
  Array res(*this);
  res.AddPos(key, pos);
  return res;
}

// 10. Удаление позиции pos (меняет текущий)
void Array::DelPos(int pos) {
  if (pos < 0 || pos >= n) return;
  if (n == 1) {
    delete[] arr; arr = nullptr; n = 0;
    return;
  }
  int* temp = new int[n - 1];
  int j = 0;
  for (int i = 0; i < n; i++) {
    if (i != pos) temp[j++] = arr[i];
  }
  delete[] arr;
  arr = temp;
  n--;
}

// Удаление позиции pos (создается новый объект)
Array Array::DelPosNew(int pos) const {
  if (pos < 0 || pos >= n) return *this;
  Array res(n - 1);
  int j = 0;
  for (int i = 0; i < n; i++) {
    if (i != pos) res.arr[j++] = arr[i];
  }
  return res;
}
11-17. Перегрузка операторов Array
// 11. Оператор присвоения (=)
Array& Array::operator=(const Array& other) {
  if (this != &other) {
    delete[] arr;
    n = other.n;
    if (n > 0) {
      arr = new int[n];
      for (int i = 0; i < n; i++) arr[i] = other.arr[i];
    } else arr = nullptr;
  }
  return *this;
}

// 12. Индексация ([])
int& Array::operator[](int index) {
  return arr[index];
}

// 13. += и + (добавление ключа key в конец)
Array& Array::operator+=(int key) {
  int* temp = new int[n + 1];
  for (int i = 0; i < n; i++) temp[i] = arr[i];
  temp[n] = key;
  delete[] arr;
  arr = temp;
  n++;
  return *this;
}
Array Array::operator+(int key) const {
  Array res(*this);
  res += key;
  return res;
}

// 14. += и + (добавление массива Array в конец)
Array& Array::operator+=(const Array& other) {
  if (other.n == 0) return *this;
  int* temp = new int[n + other.n];
  for (int i = 0; i < n; i++) temp[i] = arr[i];
  for (int i = 0; i < other.n; i++) temp[n + i] = other.arr[i];
  delete[] arr;
  arr = temp;
  n += other.n;
  return *this;
}
Array Array::operator+(const Array& other) const {
  Array res(*this);
  res += other;
  return res;
}

// 15. -= и - (удаление первого вхождения ключа key)
Array& Array::operator-=(int key) {
  int pos = FindKey(key);
  if (pos != -1) DelPos(pos);
  return *this;
}
Array Array::operator-(int key) const {
  Array res(*this);
  res -= key;
  return res;
}

// 16. Сравнение (== и !=)
bool Array::operator==(const Array& other) const {
  if (n != other.n) return false;
  for (int i = 0; i < n; i++)
    if (arr[i] != other.arr[i]) return false;
  return true;
}
bool Array::operator!=(const Array& other) const {
  return !(*this == other);
}

// 17. Потоковый ввод >> и вывод <<

  for (int i = 0; i < a.n; i++) in >> a.arr[i];
  return in;
}
ostream& operator<<(ostream& out, const Array& a) {
  for (int i = 0; i < a.n; i++) out << a.arr[i] << " ";
  return out;
}

6. Stack<T> (Шаблон стека)
Объявление шаблона Stack<T>
#include <iostream>
using namespace std;

template <class T>
class Stack {
private:
  T* arr;  // массив элементов
  int max; // размер массива
  int n;   // кол-во реальных элементов

public:
  Stack(int k = 50);
  Stack(const Stack<T>& other);
  ~Stack();

  bool IsEmpty() const;     // проверка на пустоту
  bool IsFull() const;      // проверка на полноту
  bool Push(const T& item); // добавить в стек
  bool Pop(T& item);        // удалить из стека
  T Top() const;            // вершина стека

  Stack<T>& operator=(const Stack<T>& other);

  template <class U>
  friend ostream& operator<<(ostream& out, const Stack<U>& s);
};
Методы шаблона вне класса
// 1. Конструктор
template <class T>
Stack<T>::Stack(int k) {
  max = (k > 0) ? k : 50;
  n = 0;
  arr = new T[max];
}

// 2. Конструктор копирования
template <class T>
Stack<T>::Stack(const Stack<T>& other) {
  max = other.max;
  n = other.n;
  arr = new T[max];
  for (int i = 0; i < n; i++) arr[i] = other.arr[i];
}

// 3. Деструктор
template <class T>
Stack<T>::~Stack() {
  delete[] arr;
}

// 4. Проверка на пустоту
template <class T>
bool Stack<T>::IsEmpty() const {
  return n == 0;
}

// 5. Проверка на заполненность
template <class T>
bool Stack<T>::IsFull() const {
  return n == max;
}

// 6. Добавить элемент (Push)
template <class T>
bool Stack<T>::Push(const T& item) {
  if (IsFull()) return false;
  arr[n++] = item;
  return true;
}

// 7. Удалить элемент (Pop)
template <class T>
bool Stack<T>::Pop(T& item) {
  if (IsEmpty()) return false;
  item = arr[--n];
  return true;
}

// 8. Значение с вершины (Top)
template <class T>
T Stack<T>::Top() const {
  return arr[n - 1];
}

// 9. Оператор присвоения (=)
template <class T>
Stack<T>& Stack<T>::operator=(const Stack<T>& other) {
  if (this != &other) {
    delete[] arr;
    max = other.max;
    n = other.n;
    arr = new T[max];
    for (int i = 0; i < n; i++) arr[i] = other.arr[i];
  }
  return *this;
}

// 10. Потоковый вывод <<
template <class U>
ostream& operator<<(ostream& out, const Stack<U>& s) {
  for (int i = 0; i < s.n; i++) out << s.arr[i] << " ";
  return out;
}
